**src/hooks/useDataFreshness.ts — create new file:**

This hook accepts a `dataUpdatedAt` timestamp (milliseconds, from TanStack Query's `useQuery` result) and returns a human-readable label. It re-computes the label every 60 seconds using `setInterval`.

```typescript
import { useState, useEffect } from 'react'

/**
 * FR-12 freshness thresholds:
 * 0–59s  → "just now"
 * 1–59m  → "X min ago"
 * 60m+   → "over an hour ago"
 */
function computeLabel(updatedAt: number | undefined): string {
  if (updatedAt == null) return ''
  const elapsedMs = Date.now() - updatedAt
  const seconds = Math.floor(elapsedMs / 1000)
  if (seconds < 60) return 'just now'
  const minutes = Math.floor(seconds / 60)
  if (minutes < 60) return `${minutes} min ago`
  return 'over an hour ago'
}

/**
 * Returns a human-readable freshness label for a given dataUpdatedAt timestamp.
 * Updates every 60 seconds. Returns empty string when no timestamp is provided.
 */
export function useDataFreshness(dataUpdatedAt: number | undefined): string {
  const [label, setLabel] = useState(() => computeLabel(dataUpdatedAt))

  useEffect(() => {
    // Recompute immediately when dataUpdatedAt changes (e.g., data just loaded)
    setLabel(computeLabel(dataUpdatedAt))

    if (dataUpdatedAt == null) return

    // Update every 60 seconds so "just now" → "1 min ago" → "2 min ago" etc.
    const interval = setInterval(() => {
      setLabel(computeLabel(dataUpdatedAt))
    }, 60_000)

    return () => clearInterval(interval)
  }, [dataUpdatedAt])

  return label
}
```

**src/hooks/useRecentLocations.ts — create new file:**

Manages the `recentLocations` localStorage array. Uses `RecentLocation` type from `src/types/location.ts` (already has `searchedAt: number` field).

```typescript
import { useState, useCallback } from 'react'
import type { RecentLocation } from '../types/location'

const STORAGE_KEY = 'recentLocations'
const MAX_LOCATIONS = 5

function readFromStorage(): RecentLocation[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY)
    if (!raw) return []
    const parsed = JSON.parse(raw)
    if (!Array.isArray(parsed)) return []
    return parsed as RecentLocation[]
  } catch {
    // Corrupted value or localStorage unavailable — clear and start fresh
    try { localStorage.removeItem(STORAGE_KEY) } catch { /* ignore */ }
    return []
  }
}

function writeToStorage(locations: RecentLocation[]): void {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(locations))
  } catch {
    // localStorage unavailable — in-session only, no crash
  }
}

/**
 * FR-12 recent locations management:
 * - Persists up to 5 locations in localStorage['recentLocations']
 * - Deduplicates by name (case-insensitive): existing entry is moved to front
 * - On overflow (>5): oldest entry (last in array) is dropped
 * - Graceful degradation: if localStorage unavailable, works in-session only
 */
export function useRecentLocations() {
  const [locations, setLocations] = useState<RecentLocation[]>(readFromStorage)

  const addLocation = useCallback((location: Omit<RecentLocation, 'searchedAt'>) => {
    setLocations(prev => {
      // Remove any existing entry with same name (case-insensitive deduplication)
      const filtered = prev.filter(
        loc => loc.name.toLowerCase() !== location.name.toLowerCase()
      )
      // Prepend new entry with current timestamp
      const updated: RecentLocation[] = [
        { ...location, searchedAt: Date.now() },
        ...filtered,
      ].slice(0, MAX_LOCATIONS)  // Enforce max 5

      writeToStorage(updated)
      return updated
    })
  }, [])

  const clear = useCallback(() => {
    setLocations([])
    try { localStorage.removeItem(STORAGE_KEY) } catch { /* ignore */ }
  }, [])

  return { locations, addLocation, clear }
}
```

**src/components/FreshnessIndicator.tsx — create new file:**

```tsx
import { useDataFreshness } from '../hooks/useDataFreshness'

interface FreshnessIndicatorProps {
  dataUpdatedAt: number | undefined
}

/**
 * Shows "Updated just now", "Updated 5 min ago", or "Updated over an hour ago".
 * Renders nothing if dataUpdatedAt is undefined (data not yet loaded).
 * No animation — a static text label, no motion concerns.
 */
export default function FreshnessIndicator({ dataUpdatedAt }: FreshnessIndicatorProps) {
  const label = useDataFreshness(dataUpdatedAt)

  if (!label) return null

  return (
    <p className="text-xs text-slate-400 text-right" aria-live="polite">
      Updated {label}
    </p>
  )
}
```

**src/components/RecentLocationChips.tsx — create new file:**

```tsx
import type { RecentLocation } from '../types/location'
import type { Location } from '../types/location'

interface RecentLocationChipsProps {
  locations: RecentLocation[]
  onSelect: (location: Location) => void
}

/**
 * Renders a row of tap-to-select chip buttons for recent locations.
 * Only rendered when locations.length > 0.
 * No animation on chip appearance — instant render respects prefers-reduced-motion.
 */
export default function RecentLocationChips({ locations, onSelect }: RecentLocationChipsProps) {
  if (locations.length === 0) return null

  return (
    <div className="flex flex-wrap gap-2" role="group" aria-label="Recent locations">
      {locations.map(loc => (
        <button
          key={`${loc.name}-${loc.lat}-${loc.lon}`}
          onClick={() => onSelect(loc)}
          className="
            px-3 py-1.5
            bg-slate-700 hover:bg-slate-600
            border border-slate-600 hover:border-slate-500
            rounded-full
            text-xs text-slate-300 hover:text-white
            transition-colors
            min-h-[36px]
            focus:outline-none focus:ring-2 focus:ring-blue-500
          "
          aria-label={`Load weather for ${loc.name}${loc.country ? `, ${loc.country}` : ''}`}
        >
          {loc.name}{loc.country ? `, ${loc.country}` : ''}
        </button>
      ))}
    </div>
  )
}
```

Chips are supplemental navigation; however, if the Phase 4 accessibility audit flags them, they can be upgraded to `min-h-[44px]` at that time. For now, use 36px to match the existing SettingsBar button height and keep the chips visually compact.

```tsx
import { useState } from 'react'
import CurrentConditions from './components/CurrentConditions'
import DailyForecast from './components/DailyForecast'
import FreshnessIndicator from './components/FreshnessIndicator'
import HourlyForecast from './components/HourlyForecast'
import LocationBar from './components/LocationBar'
import RecentLocationChips from './components/RecentLocationChips'
import SettingsBar from './components/SettingsBar'
import WeatherDetails from './components/WeatherDetails'
import { useWeather } from './hooks/useWeather'
import { useRecentLocations } from './hooks/useRecentLocations'
import { getWeatherCondition } from './utils/weatherCodes'
import type { Location } from './types/location'

export default function AppShell() {
  const [activeLocation, setActiveLocation] = useState<Location | null>(null)
  const { locations: recentLocations, addLocation } = useRecentLocations()

  // Reuses TanStack Query cache — no extra network call
  const { data: weatherData, dataUpdatedAt } = useWeather(
    activeLocation?.lat ?? null,
    activeLocation?.lon ?? null
  )

  // Derive background gradient class from live weather data
  const bgClass = (() => {
    if (!activeLocation || !weatherData) return 'bg-slate-900'
    const condition = getWeatherCondition(weatherData.current.weatherCode)
    const isDay = weatherData.current.isDay === 1
    const gradientClass = isDay
      ? condition.backgroundClass
      : condition.backgroundClassNight
    return `bg-gradient-to-br ${gradientClass}`
  })()

  // Called whenever a location is selected (from search, GPS, or chip)
  function handleLocationSelect(location: Location) {
    setActiveLocation(location)
    addLocation(location)
  }

  return (
    <div className={`min-h-screen ${bgClass} text-white flex flex-col`}>
      <div className="max-w-2xl w-full mx-auto px-4 py-8 space-y-4 flex-1">
        {/* Location search + GPS */}
        <LocationBar
          onLocationSelect={handleLocationSelect}
          currentLocation={activeLocation}
        />

        {/* Recent location chips — only shown when there are recent locations */}
        <RecentLocationChips
          locations={recentLocations}
          onSelect={handleLocationSelect}
        />

        {/* °C / °F toggle + freshness indicator */}
        <div className="flex items-center justify-between gap-4">
          <FreshnessIndicator dataUpdatedAt={dataUpdatedAt} />
          <SettingsBar />
        </div>

        {/* Weather content */}
        {activeLocation ? (
          <div className="space-y-4">
            <CurrentConditions location={activeLocation} />
            <HourlyForecast lat={activeLocation.lat} lon={activeLocation.lon} />
            <DailyForecast lat={activeLocation.lat} lon={activeLocation.lon} />
            <WeatherDetails lat={activeLocation.lat} lon={activeLocation.lon} />
          </div>
        ) : (
          <div className="text-center text-slate-400 py-20">
            <p className="text-xl font-medium mb-2">Simple Weather</p>
            <p className="text-sm">Search for a city or use your location to get started</p>
          </div>
        )}
      </div>

      {/* Attribution footer — Open-Meteo CC BY 4.0 required */}
      <footer className="text-center text-slate-500 text-xs py-4">
        Weather data from{' '}
        <a
          href="https://open-meteo.com"
          target="_blank"
          rel="noopener noreferrer"
          className="underline hover:text-slate-300 transition-colors"
        >
          Open-Meteo
        </a>
        {' '}(CC BY 4.0)
      </footer>
    </div>
  )
}
```

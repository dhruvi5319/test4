**src/components/WeatherDetails.tsx — create new file:**

```tsx
import { useState } from 'react'
import { ChevronDown, ChevronUp } from 'lucide-react'
import { useWeather } from '../hooks/useWeather'
import { useUnitPreference } from '../hooks/useUnitPreference'
import { degreesToCardinal, formatLocalTime } from '../utils/formatters'

interface WeatherDetailsProps {
  lat: number
  lon: number
}

/** UV index integer → severity label per FR-08 spec */
function uvLabel(uvi: number): string {
  if (uvi <= 2) return 'Low'
  if (uvi <= 5) return 'Moderate'
  if (uvi <= 7) return 'High'
  if (uvi <= 10) return 'Very High'
  return 'Extreme'
}

/** Visibility in metres → human-readable string per unit system */
function formatVisibility(metres: number, unit: 'metric' | 'imperial'): string {
  if (unit === 'metric') {
    return `${Math.round(metres / 1000)} km`
  }
  return `${Math.round(metres / 1609.34)} mi`
}

export default function WeatherDetails({ lat, lon }: WeatherDetailsProps) {
  const [expanded, setExpanded] = useState(false)
  const { data } = useWeather(lat, lon)
  const { unit } = useUnitPreference()

  // Only render if data is available — no skeleton needed (panel is collapsed by default)
  if (!data) return null

  const current = data.current
  const daily0 = data.daily[0]

  const uvIndex = daily0?.uvIndexMax != null ? Math.round(daily0.uvIndexMax) : null
  const windDir = degreesToCardinal(current.windDirection)
  const visibility = current.visibility != null ? formatVisibility(current.visibility, unit) : '—'
  const sunrise = daily0?.sunrise ? formatLocalTime(daily0.sunrise, data.timezone) : '—'
  const sunset = daily0?.sunset ? formatLocalTime(daily0.sunset, data.timezone) : '—'

  return (
    <div className="rounded-2xl bg-slate-800 overflow-hidden">
      {/* Toggle button */}
      <button
        onClick={() => setExpanded(prev => !prev)}
        aria-expanded={expanded}
        aria-controls="weather-details-panel"
        className="w-full flex items-center justify-between px-6 py-4 text-slate-300 hover:text-white hover:bg-slate-700 transition-colors focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-inset min-h-[44px]"
      >
        <span className="text-sm font-medium">Details</span>
        {expanded ? <ChevronUp size={18} aria-hidden="true" /> : <ChevronDown size={18} aria-hidden="true" />}
      </button>

      {/* Collapsible panel — no height animation; instant show/hide respects prefers-reduced-motion */}
      {expanded && (
        <div
          id="weather-details-panel"
          className="px-6 pb-5 pt-1 grid grid-cols-2 gap-x-4 gap-y-4 border-t border-slate-700"
        >
          {/* UV Index */}
          {uvIndex != null && (
            <div className="flex flex-col gap-0.5">
              <span className="text-xs text-slate-400">UV Index</span>
              <span className="text-white font-medium text-sm">
                {uvIndex} — {uvLabel(uvIndex)}
              </span>
            </div>
          )}

          {/* Wind Direction */}
          <div className="flex flex-col gap-0.5">
            <span className="text-xs text-slate-400">Wind Direction</span>
            <span className="text-white font-medium text-sm">{windDir}</span>
          </div>

          {/* Visibility */}
          <div className="flex flex-col gap-0.5">
            <span className="text-xs text-slate-400">Visibility</span>
            <span className="text-white font-medium text-sm">{visibility}</span>
          </div>

          {/* Sunrise */}
          <div className="flex flex-col gap-0.5">
            <span className="text-xs text-slate-400">Sunrise</span>
            <span className="text-white font-medium text-sm">{sunrise}</span>
          </div>

          {/* Sunset */}
          <div className="flex flex-col gap-0.5">
            <span className="text-xs text-slate-400">Sunset</span>
            <span className="text-white font-medium text-sm">{sunset}</span>
          </div>
        </div>
      )}
    </div>
  )
}
```

**Why instant show/hide instead of CSS transition:**
FR-08 requires `prefers-reduced-motion` compliance. Using a conditional render (`{expanded && ...}`) provides an instant toggle that always respects the media query without needing to detect it explicitly. A CSS `transition` on `max-height` would require a `prefers-reduced-motion` media query check. Instant render is simpler and always correct.

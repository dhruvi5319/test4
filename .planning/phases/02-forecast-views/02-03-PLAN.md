**4. Full component:**
```tsx
import {
  AreaChart,
  Area,
  XAxis,
  YAxis,
  Tooltip,
  ResponsiveContainer,
  CartesianGrid,
} from 'recharts'
import { formatLocalDay } from '../utils/formatters'
import type { DailyWeather, UnitSystem } from '../types/weather'

export default function TemperatureTrendChart({
  data,
  timezone,
  unit,
}: TemperatureTrendChartProps) {
  const prefersReducedMotion =
    typeof window !== 'undefined' &&
    window.matchMedia('(prefers-reduced-motion: reduce)').matches

  const yUnit = unit === 'metric' ? '°C' : '°F'

  const chartData = data.map((day, i) => ({
    day: formatLocalDay(day.date, timezone, i),
    high: day.tempMax != null ? Math.round(day.tempMax) : null,
    low: day.tempMin != null ? Math.round(day.tempMin) : null,
  }))

  return (
    <div className="mt-4" role="img" aria-label="Temperature trend chart for the next 7 days">
      {/* Visual chart */}
      <ResponsiveContainer width="100%" height={160}>
        <AreaChart data={chartData} margin={{ top: 8, right: 8, left: -16, bottom: 0 }}>
          <defs>
            <linearGradient id="highGrad" x1="0" y1="0" x2="0" y2="1">
              <stop offset="5%" stopColor="#f97316" stopOpacity={0.4} />
              <stop offset="95%" stopColor="#f97316" stopOpacity={0} />
            </linearGradient>
            <linearGradient id="lowGrad" x1="0" y1="0" x2="0" y2="1">
              <stop offset="5%" stopColor="#60a5fa" stopOpacity={0.4} />
              <stop offset="95%" stopColor="#60a5fa" stopOpacity={0} />
            </linearGradient>
          </defs>

          <CartesianGrid strokeDasharray="3 3" stroke="#334155" vertical={false} />

          <XAxis
            dataKey="day"
            tick={{ fontSize: 11, fill: '#94a3b8' }}
            axisLine={false}
            tickLine={false}
          />

          <YAxis
            tick={{ fontSize: 11, fill: '#94a3b8' }}
            axisLine={false}
            tickLine={false}
            tickFormatter={v => `${v}${yUnit}`}
            width={44}
          />

          <Tooltip
            contentStyle={{ backgroundColor: '#1e293b', border: '1px solid #334155', borderRadius: '8px' }}
            labelStyle={{ color: '#94a3b8', fontSize: 12 }}
            itemStyle={{ color: '#e2e8f0', fontSize: 12 }}
            formatter={(value: number, name: string) => [`${Math.round(value)}${yUnit}`, name === 'high' ? 'High' : 'Low']}
          />

          {/* High temp area — warm orange */}
          <Area
            type="monotone"
            dataKey="high"
            stroke="#f97316"
            strokeWidth={2}
            fill="url(#highGrad)"
            dot={false}
            isAnimationActive={!prefersReducedMotion}
            name="high"
          />

          {/* Low temp area — cool blue */}
          <Area
            type="monotone"
            dataKey="low"
            stroke="#60a5fa"
            strokeWidth={2}
            fill="url(#lowGrad)"
            dot={false}
            isAnimationActive={!prefersReducedMotion}
            name="low"
          />
        </AreaChart>
      </ResponsiveContainer>

      {/* Screen reader fallback — visually hidden (WCAG + FR-11 + TECH-ARCH §12) */}
      <table className="sr-only">
        <caption>7-day temperature trend</caption>
        <thead>
          <tr>
            <th scope="col">Day</th>
            <th scope="col">High</th>
            <th scope="col">Low</th>
          </tr>
        </thead>
        <tbody>
          {chartData.map(row => (
            <tr key={row.day}>
              <td>{row.day}</td>
              <td>{row.high != null ? `${row.high}${yUnit}` : '—'}</td>
              <td>{row.low != null ? `${row.low}${yUnit}` : '—'}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}
```

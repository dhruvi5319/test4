**src/components/SettingsBar.tsx** — Toggle button with visual highlight on active unit. Reads/writes via `useUnitPreference` (already wraps localStorage in try/catch — no additional error handling needed here).

FR-04 validation rules:
- Only two valid values: `'metric'` or `'imperial'`
- Default to `'metric'` (handled in useUnitPreference)
- Toggle is always visible on the main screen — never hidden (per FRD FR-04)
- `role="switch"` for ARIA (from TECH-ARCH.md section 12)
- `aria-checked` reflects imperial=true, metric=false

```typescript
import { useUnitPreference } from '../hooks/useUnitPreference';

export function SettingsBar() {
  const { unit, setUnit } = useUnitPreference();

  const toggle = () => setUnit(unit === 'metric' ? 'imperial' : 'metric');

  return (
    <div className="flex justify-end">
      <button
        role="switch"
        aria-checked={unit === 'imperial'}
        aria-label="Temperature unit"
        onClick={toggle}
        className="flex items-center gap-0.5 rounded-full p-1 bg-white/10 border border-white/20 hover:bg-white/20 transition-colors"
      >
        <span
          className={`px-3 py-1 rounded-full text-sm font-medium transition-colors ${
            unit === 'metric'
              ? 'bg-white text-slate-900'
              : 'text-slate-300'
          }`}
        >
          °C
        </span>
        <span
          className={`px-3 py-1 rounded-full text-sm font-medium transition-colors ${
            unit === 'imperial'
              ? 'bg-white text-slate-900'
              : 'text-slate-300'
          }`}
        >
          °F
        </span>
      </button>
    </div>
  );
}
```

```typescript
import { test, expect } from '@playwright/test';

test.describe('Unit Toggle (°C/°F)', () => {
  test.beforeEach(async ({ page }) => {
    // Clear localStorage before each test to start with clean state
    await page.goto('/');
    await page.evaluate(() => localStorage.clear());
    await page.reload();
  });

  test('°C is the default unit when no localStorage value exists', async ({ page }) => {
    // After clearing localStorage and reloading, toggle should show °C as active
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    await expect(toggle).toBeVisible();
    // °C span should have the active (white background) styling
    const celsiusSpan = toggle.getByText('°C');
    await expect(celsiusSpan).toBeVisible();
    // Check aria-checked is false (not imperial)
    await expect(toggle).toHaveAttribute('aria-checked', 'false');
  });

  test('clicking the toggle switches from °C to °F', async ({ page }) => {
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    await expect(toggle).toHaveAttribute('aria-checked', 'false'); // metric
    await toggle.click();
    await expect(toggle).toHaveAttribute('aria-checked', 'true'); // imperial
  });

  test('unit preference persists after page reload', async ({ page }) => {
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    // Switch to °F
    await toggle.click();
    await expect(toggle).toHaveAttribute('aria-checked', 'true');
    // Reload the page
    await page.reload();
    // °F should still be active
    const reloadedToggle = page.getByRole('switch', { name: /temperature unit/i });
    await expect(reloadedToggle).toHaveAttribute('aria-checked', 'true');
  });

  test('localStorage key is weatherUnit with value metric or imperial', async ({ page }) => {
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    // Default: metric
    const initialValue = await page.evaluate(() => localStorage.getItem('weatherUnit'));
    // It may be null (not set yet) or 'metric'
    expect(initialValue === null || initialValue === 'metric').toBe(true);
    // Switch to imperial
    await toggle.click();
    const imperialValue = await page.evaluate(() => localStorage.getItem('weatherUnit'));
    expect(imperialValue).toBe('imperial');
    // Switch back to metric
    await toggle.click();
    const metricValue = await page.evaluate(() => localStorage.getItem('weatherUnit'));
    expect(metricValue).toBe('metric');
  });

  test('unit toggle is always visible (never hidden)', async ({ page }) => {
    // Visible before location selected
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    await expect(toggle).toBeVisible();
    // Also visible after selecting a location
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Madrid');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();
    // Toggle still visible after location selected
    await expect(toggle).toBeVisible();
  });

  test('complete Phase 1 flow: search → weather loads → toggle unit → temperatures update', async ({ page }) => {
    // Step 1: Search for a city
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Sydney');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    // Step 2: Wait for weather data to load (skeleton disappears, real data appears)
    const weatherSection = page.getByRole('region', { name: /current weather/i });
    await expect(weatherSection).toBeVisible({ timeout: 10000 });

    // Step 3: Read the current temperature in °C
    const tempElement = weatherSection.locator('.text-7xl');
    const celsiusTemp = await tempElement.textContent();
    expect(celsiusTemp).toMatch(/°C$/);

    // Step 4: Toggle to °F
    const toggle = page.getByRole('switch', { name: /temperature unit/i });
    await toggle.click();

    // Step 5: Wait for °F data to load and verify temperature changed
    await expect(weatherSection).toBeVisible({ timeout: 10000 });
    const fahrenheitTemp = await tempElement.textContent();
    expect(fahrenheitTemp).toMatch(/°F$/);

    // Step 6: Temperature value should be different (F ≠ C for most temperatures)
    // (unless it's exactly -40, where C and F are equal)
    expect(celsiusTemp).not.toBe(fahrenheitTemp);
  });
});

test.describe('No blank screen rule (FR-10)', () => {
  test('page never shows a blank screen on initial load', async ({ page }) => {
    await page.goto('/');
    // Some content must always be visible
    const bodyText = await page.locator('body').textContent();
    expect(bodyText?.trim().length).toBeGreaterThan(0);
    // The "search for a city" message is the initial non-blank state
    await expect(page.getByText(/search for a city/i)).toBeVisible();
  });

  test('Open-Meteo attribution footer is visible', async ({ page }) => {
    await page.goto('/');
    await expect(page.getByText(/open-meteo/i)).toBeVisible();
  });
});
```

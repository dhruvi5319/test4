Locked decisions to implement:
- Suggestions appear after 2+ chars (enforced by useGeocode's `enabled` guard — no extra check needed here)
- Dropdown: simple list below input — `{name}, {admin1 ? admin1 + ', ' : ''}{country_code}` per RESEARCH.md disambiguation recommendation

- Enter key: selects first suggestion

Claude's discretion choices:

- GPS loading: spinner icon while geolocating
- Keyboard: arrow keys for navigation
- Clear (×) button when input has text

```typescript
import { useEffect, useRef, useState } from 'react';
import { MapPin, Search, X, Loader2 } from 'lucide-react';
import { useGeocode } from '../hooks/useGeocode';
import { reverseGeocode } from '../api/nominatim';
import type { GeocodingResult } from '../types/openMeteoRaw';
import type { Location } from '../types/location';

interface LocationBarProps {
  onLocationSelect: (location: Location) => void;
}

export function LocationBar({ onLocationSelect }: LocationBarProps) {
  const [inputValue, setInputValue] = useState('');
  const [isDropdownOpen, setIsDropdownOpen] = useState(false);
  const [isGeolocating, setIsGeolocating] = useState(false);
  const [gpsError, setGpsError] = useState<string | null>(null);
  const [highlightedIndex, setHighlightedIndex] = useState(-1);
  const inputRef = useRef<HTMLInputElement>(null);
  const dropdownRef = useRef<HTMLUListElement>(null);

  // useGeocode handles 2+ char enforcement and debouncing internally
  const { data: suggestions = [], isLoading: isFetchingSuggestions, isError: isGeocodeError } =
    useGeocode(inputValue);

  // Show dropdown when we have results or are loading (only for 2+ char input)
  useEffect(() => {
    setIsDropdownOpen(
      inputValue.length >= 2 && (suggestions.length > 0 || isFetchingSuggestions)
    );
    setHighlightedIndex(-1);
  }, [inputValue, suggestions, isFetchingSuggestions]);

  const handleSelect = (result: GeocodingResult) => {
    const displayName = `${result.name}${result.admin1 ? `, ${result.admin1}` : ''}, ${result.country_code}`;
    setInputValue(result.name); // show city name in input (locked decision)
    setIsDropdownOpen(false);   // close dropdown (locked decision)
    setGpsError(null);
    onLocationSelect({
      lat: result.latitude,
      lon: result.longitude,
      name: displayName,
      country: result.country_code,
      admin1: result.admin1,
    });
  };

  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (!isDropdownOpen) return;

    if (e.key === 'Enter') {
      // Enter selects first suggestion (locked decision) or highlighted item
      const targetIndex = highlightedIndex >= 0 ? highlightedIndex : 0;
      if (suggestions[targetIndex]) {
        e.preventDefault();
        handleSelect(suggestions[targetIndex]);
      }
    } else if (e.key === 'Escape') {
      setIsDropdownOpen(false); // locked decision: Escape closes
    } else if (e.key === 'ArrowDown') {
      e.preventDefault();
      setHighlightedIndex(prev => Math.min(prev + 1, suggestions.length - 1));
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setHighlightedIndex(prev => Math.max(prev - 1, -1));
    }
  };

  // GPS handler — NEVER called automatically; only called on button click
  function handleGpsClick() {
    if (!navigator.geolocation) {
      setGpsError('Geolocation is not supported by your browser');
      return;
    }
    setIsGeolocating(true);
    setGpsError(null);

    navigator.geolocation.getCurrentPosition(
      async (position) => {
        try {
          const { latitude, longitude } = position.coords;
          const name = await reverseGeocode(latitude, longitude);
          setInputValue(name);
          onLocationSelect({ lat: latitude, lon: longitude, name });
        } catch {
          setGpsError('Unable to determine your city name — try searching instead');
        } finally {
          setIsGeolocating(false);
        }
      },
      (error) => {
        setIsGeolocating(false);
        // All 3 error codes handled — locked decision: no blank screen, no stuck state
        switch (error.code) {
          case error.PERMISSION_DENIED:
            setGpsError('Location access denied — search for a city above');
            break;
          case error.POSITION_UNAVAILABLE:
            setGpsError('Unable to detect location — search for a city above');
            break;
          case error.TIMEOUT:
            setGpsError('Location timed out — search for a city above');
            break;
          default:
            setGpsError('Location error — search for a city above');
        }
      },
      { timeout: 10000, maximumAge: 300000 }
    );
  }

  // Close dropdown when clicking outside
  useEffect(() => {
    function handleOutsideClick(e: MouseEvent) {
      if (
        inputRef.current && !inputRef.current.contains(e.target as Node) &&
        dropdownRef.current && !dropdownRef.current.contains(e.target as Node)
      ) {
        setIsDropdownOpen(false);
      }
    }
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  return (
    <div className="relative w-full">
      {/* Search input row */}
      <div className="flex gap-2">
        <div className="relative flex-1">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
          <input
            ref={inputRef}
            type="text"
            role="combobox"
            aria-autocomplete="list"
            aria-controls="location-suggestions"
            aria-expanded={isDropdownOpen}
            aria-label="Search for a city"
            value={inputValue}
            onChange={e => {
              setInputValue(e.target.value);
              setGpsError(null);
            }}
            onKeyDown={handleKeyDown}
            placeholder="Search city..."
            className="w-full pl-9 pr-8 py-3 rounded-xl bg-white/10 text-white placeholder-slate-400 border border-white/20 focus:outline-none focus:ring-2 focus:ring-white/40"
          />
          {inputValue && (
            <button
              onClick={() => { setInputValue(''); setIsDropdownOpen(false); }}
              aria-label="Clear search"
              className="absolute right-2 top-1/2 -translate-y-1/2 p-1 rounded-full hover:bg-white/10 text-slate-400"
            >
              <X className="w-3 h-3" />
            </button>
          )}
        </div>

        {/* GPS button — opt-in only, never auto-triggered */}
        <button
          onClick={handleGpsClick}
          disabled={isGeolocating}
          aria-label="Use my current location"
          aria-pressed={isGeolocating}
          className="p-3 rounded-xl bg-white/10 border border-white/20 hover:bg-white/20 disabled:opacity-50 min-w-[48px] min-h-[48px] flex items-center justify-center"
        >
          {isGeolocating
            ? <Loader2 className="w-5 h-5 text-white animate-spin" />
            : <MapPin className="w-5 h-5 text-white" />
          }
        </button>
      </div>

      {/* GPS error message — inline, search remains usable */}
      {gpsError && (
        <p role="alert" className="mt-1 text-sm text-red-300 px-1">{gpsError}</p>
      )}

      {/* Geocoding error message */}
      {isGeocodeError && inputValue.length >= 2 && (
        <p role="alert" className="mt-1 text-sm text-red-300 px-1">
          Unable to search — check your connection
        </p>
      )}

      {/* Autocomplete dropdown — simple list (locked decision: no extra decoration) */}
      {isDropdownOpen && (
        <ul
          id="location-suggestions"
          ref={dropdownRef}
          role="listbox"
          aria-label="City suggestions"
          className="absolute top-full left-0 right-0 mt-1 bg-white rounded-xl shadow-lg border border-slate-200 z-50 overflow-hidden"
        >
          {isFetchingSuggestions && suggestions.length === 0 && (
            <li className="px-4 py-3 text-slate-500 text-sm flex items-center gap-2">
              <Loader2 className="w-4 h-4 animate-spin" /> Searching...
            </li>
          )}
          {!isFetchingSuggestions && suggestions.length === 0 && inputValue.length >= 2 && (
            <li className="px-4 py-3 text-slate-500 text-sm">
              No cities found — try a different spelling
            </li>
          )}
          {suggestions.map((result, index) => {
            const displayName = `${result.name}${result.admin1 ? `, ${result.admin1}` : ''}, ${result.country_code}`;
            return (
              <li
                key={result.id}
                role="option"
                aria-selected={index === highlightedIndex}
                onClick={() => handleSelect(result)}
                className={`px-4 py-3 cursor-pointer text-slate-800 text-sm hover:bg-slate-100 ${
                  index === highlightedIndex ? 'bg-slate-100' : ''
                }`}
              >
                {displayName}
              </li>
            );
          })}
        </ul>
      )}
    </div>
  );
}
```

```typescript
import { test, expect } from '@playwright/test';

test.describe('LocationBar', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/');
  });

  test('shows no dropdown for 0-1 character input', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('L');
    // Wait to confirm dropdown does NOT appear (even after debounce fires)
    await page.waitForTimeout(500);
    await expect(page.getByRole('listbox')).not.toBeVisible();
  });

  test('shows dropdown with suggestions after 2+ characters', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    // Mock network or use real API — type "Lo" and wait for dropdown
    await input.fill('Lo');
    // Wait for debounce (350ms) + network response
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    const options = page.getByRole('option');
    await expect(options).toHaveCount({ minimum: 1 });
  });

  test('selecting a suggestion shows city name in input and closes dropdown', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('London');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    const firstOption = page.getByRole('option').first();
    const cityName = await firstOption.textContent();
    await firstOption.click();
    // Dropdown closes
    await expect(page.getByRole('listbox')).not.toBeVisible();
    // Input shows city name
    const inputValue = await input.inputValue();
    expect(inputValue.length).toBeGreaterThan(0);
  });

  test('pressing Enter selects the first suggestion', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Paris');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await input.press('Enter');
    // Dropdown closes after Enter
    await expect(page.getByRole('listbox')).not.toBeVisible();
    // Input has a value (city name was populated)
    const inputValue = await input.inputValue();
    expect(inputValue.length).toBeGreaterThan(0);
  });

  test('pressing Escape closes the dropdown without selecting', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    const originalValue = 'Berlin';
    await input.fill(originalValue);
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await input.press('Escape');
    await expect(page.getByRole('listbox')).not.toBeVisible();
    // Input still has the typed value (not cleared)
    expect(await input.inputValue()).toBe(originalValue);
  });

  test('GPS button is always visible and does not auto-trigger geolocation', async ({ page }) => {
    // GPS button must exist on page load
    const gpsButton = page.getByRole('button', { name: /use my current location/i });
    await expect(gpsButton).toBeVisible();
    // The page should load without any geolocation permission prompt
    // (no auto-request on page load — locked decision)
    // Confirm no geolocation was requested by verifying no error state appeared
    await expect(page.getByRole('alert')).not.toBeVisible();
  });

  test('shows no-results message when city not found', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('xyzzynotacity');
    // Wait for API response
    await page.waitForTimeout(1000);
    // Either no dropdown, or dropdown with "no cities found" message
    const noResults = page.getByText(/no cities found/i);
    // If visible, check message; if dropdown never opened, that's also acceptable
    const listbox = page.getByRole('listbox');
    if (await listbox.isVisible()) {
      await expect(noResults).toBeVisible();
    }
  });

  test('clear button appears when input has text and resets the input', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Tokyo');
    const clearButton = page.getByRole('button', { name: /clear search/i });
    await expect(clearButton).toBeVisible();
    await clearButton.click();
    expect(await input.inputValue()).toBe('');
    await expect(clearButton).not.toBeVisible();
  });
});
```

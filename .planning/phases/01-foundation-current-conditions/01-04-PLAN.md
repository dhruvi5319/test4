**src/components/CurrentConditions.tsx** — Hero weather display. Uses `is_day` from API response for day/night icon variant. All temperatures use `formatTemperature` (integer display). Pattern from TECH-ARCH.md section 8: `isLoading → skeleton, isError → error card`:

Note: `is_day` is a number (0 or 1) from the raw API response. Use it directly — simpler and more reliable than computing from sunrise/sunset.

```typescript
import { Wind, Droplets, CloudRain } from 'lucide-react';
import * as LucideIcons from 'lucide-react';
import { useWeather } from '../hooks/useWeather';
import { useUnitPreference } from '../hooks/useUnitPreference';
import { getWeatherCondition } from '../utils/weatherCodes';
import { formatTemperature } from '../utils/formatters';
import { CurrentConditionsSkeleton } from './CurrentConditionsSkeleton';
import { WeatherErrorCard } from './WeatherErrorCard';

interface CurrentConditionsProps {
  lat: number;
  lon: number;
  locationName: string;
}

// Render a lucide-react icon by its string name from weatherCodes.ts
function WeatherIcon({ name, isDaytime, className }: {
  name: string;
  isDaytime: boolean;
  className?: string;
}) {
  const condition = { iconDay: name, iconNight: name };
  const iconName = isDaytime ? condition.iconDay : condition.iconNight;
  // @ts-ignore — dynamic icon lookup
  const Icon = LucideIcons[iconName] ?? LucideIcons.Cloud;
  return <Icon className={className} aria-hidden="true" />;
}

export function CurrentConditions({ lat, lon, locationName }: CurrentConditionsProps) {
  const { unit } = useUnitPreference();
  const { data, isLoading, isError, refetch } = useWeather(lat, lon);

  // isLoading = isPending && isFetching (TanStack Query v5 derived flag)
  // Safe for both initial load and lazy/disabled query states
  if (isLoading) return <CurrentConditionsSkeleton />;
  if (isError) return <WeatherErrorCard onRetry={() => refetch()} />;
  if (!data) return <CurrentConditionsSkeleton />;

  const current = data.current;
  const daily = data.daily;
  const isDaytime = current.is_day === 1;
  const condition = getWeatherCondition(current.weather_code);
  const iconName = isDaytime ? condition.iconDay : condition.iconNight;

  // Today's high and low — from daily[0]
  const todayHigh = daily?.temperature_2m_max?.[0] ?? null;
  const todayLow = daily?.temperature_2m_min?.[0] ?? null;

  // @ts-ignore — dynamic lucide icon lookup
  const ConditionIcon = LucideIcons[iconName] ?? LucideIcons.Cloud;

  return (
    <section
      aria-live="polite"
      aria-label="Current weather conditions"
      className="p-6 rounded-2xl bg-white/10 space-y-4"
    >
      {/* Location name */}
      <h2 className="text-slate-200 text-sm font-medium">{locationName}</h2>

      {/* Temperature row — large/dominant per FR-03 */}
      <div className="flex items-center justify-between">
        <div>
          <p className="text-7xl font-bold text-white leading-none">
            {formatTemperature(current.temperature_2m, unit)}
          </p>
          <p className="text-slate-300 mt-2">
            Feels like {formatTemperature(current.apparent_temperature, unit)}
          </p>
        </div>
        <ConditionIcon
          className="w-16 h-16 text-white"
          aria-label={`${condition.label} icon`}
        />
      </div>

      {/* Condition text label */}
      <p className="text-white text-lg font-medium">{condition.label}</p>

      {/* High / Low */}
      {todayHigh !== null && todayLow !== null && (
        <p className="text-slate-200 text-sm">
          H: {formatTemperature(todayHigh, unit)} · L: {formatTemperature(todayLow, unit)}
        </p>
      )}

      {/* Metrics row: humidity, wind, precip */}
      <div className="flex flex-wrap gap-4 text-sm text-slate-200">
        <span className="flex items-center gap-1">
          <Droplets className="w-4 h-4" aria-hidden="true" />
          {Math.round(current.relative_humidity_2m)}%
        </span>
        <span className="flex items-center gap-1">
          <Wind className="w-4 h-4" aria-hidden="true" />
          {Math.round(current.wind_speed_10m)}{unit === 'metric' ? ' km/h' : ' mph'}
        </span>
        <span className="flex items-center gap-1">
          <CloudRain className="w-4 h-4" aria-hidden="true" />
          {Math.round(current.precipitation_probability)}%
        </span>
      </div>
    </section>
  );
}
```

import { test, expect } from '@playwright/test';

test.describe('CurrentConditions', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/');
  });

  test('shows empty state (no weather card) before location is selected', async ({ page }) => {
    await expect(page.getByText(/search for a city to see weather/i)).toBeVisible();
    await expect(page.getByRole('region', { name: /current weather/i })).not.toBeVisible();
  });

  test('shows skeleton while weather data is loading', async ({ page }) => {
    // Select a city to trigger weather fetch
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('London');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    // Skeleton should appear immediately (before API responds)
    // Either skeleton or real data is visible — never blank
    const hasContent = await page.evaluate(() => {
      const loading = document.querySelector('[aria-label="Loading weather data"]');
      const content = document.querySelector('[aria-label="Current weather conditions"]');
      return !!(loading || content);
    });
    expect(hasContent).toBe(true);
  });

  test('displays current temperature as an integer (no decimals) after city selection', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Paris');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    // Wait for weather data to load (skeleton disappears)
    await expect(page.getByRole('region', { name: /current weather/i }))
      .toBeVisible({ timeout: 10000 });

    // Temperature text should match integer pattern: "15°C" or "59°F"
    const weatherSection = page.getByRole('region', { name: /current weather/i });
    const tempText = await weatherSection.locator('.text-7xl').textContent();
    expect(tempText).toMatch(/^-?\d+°[CF]$/);
  });

  test('displays feels-like temperature', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Tokyo');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    await expect(page.getByRole('region', { name: /current weather/i }))
      .toBeVisible({ timeout: 10000 });
    await expect(page.getByText(/feels like/i)).toBeVisible();
  });

  test('displays humidity, wind speed, and precipitation probability', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Berlin');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    await expect(page.getByRole('region', { name: /current weather/i }))
      .toBeVisible({ timeout: 10000 });

    const section = page.getByRole('region', { name: /current weather/i });
    // Check for % symbols (humidity and precip) and wind unit
    const sectionText = await section.textContent();
    expect(sectionText).toMatch(/%/); // humidity or precip %
    expect(sectionText).toMatch(/km\/h|mph/i); // wind speed unit
  });

  test('shows error card with retry button when API fails (simulated)', async ({ page }) => {
    // Block the Open-Meteo API to force an error
    await page.route('**/api.open-meteo.com/**', route => route.abort());

    const input = page.getByRole('combobox', { name: /search for a city/i });
    // We need to fill the input and simulate selection without using the geocode API
    // Use evaluate to directly set AppShell state via a custom event or localStorage trick
    // Simpler: directly navigate with mock location via URL param not supported — instead
    // intercept geocoding too and return mock data
    await page.route('**/geocoding-api.open-meteo.com/**', route => {
      route.fulfill({
        status: 200,
        contentType: 'application/json',
        body: JSON.stringify({
          results: [{
            id: 1,
            name: 'London',
            latitude: 51.5074,
            longitude: -0.1278,
            country: 'United Kingdom',
            country_code: 'GB',
            admin1: 'England',
          }]
        })
      });
    });

    await input.fill('London');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    // Error card should appear (API is blocked)
    await expect(page.getByRole('alert')).toBeVisible({ timeout: 10000 });
    await expect(page.getByRole('button', { name: /try again/i })).toBeVisible();
    // Never a blank screen
    await expect(page.getByText(/unable to load weather/i)).toBeVisible();
  });

  test('shows location name as section heading', async ({ page }) => {
    const input = page.getByRole('combobox', { name: /search for a city/i });
    await input.fill('Rome');
    await expect(page.getByRole('listbox')).toBeVisible({ timeout: 5000 });
    await page.getByRole('option').first().click();

    await expect(page.getByRole('region', { name: /current weather/i }))
      .toBeVisible({ timeout: 10000 });

    const section = page.getByRole('region', { name: /current weather/i });
    const sectionText = await section.textContent();
    expect(sectionText?.toLowerCase()).toMatch(/rome|roma/i);
  });
});

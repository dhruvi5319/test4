**Create src/components/OfflineBanner.tsx:**

New component. Listens to browser online/offline events and renders a banner when offline:

```tsx
import { useEffect, useState } from 'react'

export default function OfflineBanner() {
  const [isOnline, setIsOnline] = useState(navigator.onLine)

  useEffect(() => {
    function handleOnline() { setIsOnline(true) }
    function handleOffline() { setIsOnline(false) }

    window.addEventListener('online', handleOnline)
    window.addEventListener('offline', handleOffline)
    return () => {
      window.removeEventListener('online', handleOnline)
      window.removeEventListener('offline', handleOffline)
    }
  }, [])

  if (isOnline) return null

  return (
    <div
      role="status"
      aria-live="polite"
      className="w-full px-4 py-2 bg-amber-600/90 text-white text-sm text-center rounded-lg"
    >
      You're offline — showing last known weather
    </div>
  )
}
```

**src/components/CurrentConditions.tsx — render stale state:**

In the restructured component (from 04-01, where aria-live wraps all states):

Add a `StaleBanner` sub-component:
```tsx
function StaleBanner({ dataUpdatedAt }: { dataUpdatedAt: number }) {
  const minutesAgo = Math.round((Date.now() - dataUpdatedAt) / 60000)
  const label = minutesAgo === 0
    ? 'just now'
    : minutesAgo === 1
    ? '1 minute ago'
    : `${minutesAgo} minutes ago`

  return (
    <p className="text-xs text-amber-400 text-center py-1" role="status">
      Showing data from {label} — unable to refresh
    </p>
  )
}
```

In the main component render logic, handle the stale-data case:

```tsx
// isError=true AND data exists = stale cache scenario
{isError && data && (
  <>
    <StaleBanner dataUpdatedAt={dataUpdatedAt ?? Date.now()} />
    <CurrentConditionsContent data={data} location={location} unit={unit} />
  </>
)}
// isError=true AND no data = complete failure
{isError && !data && <WeatherErrorCard onRetry={() => refetch()} />}
// loading with no data
{isLoading && !data && <CurrentConditionsSkeleton />}
// success
{data && !isError && <CurrentConditionsContent data={data} location={location} unit={unit} />}
```
